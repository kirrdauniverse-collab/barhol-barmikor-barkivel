"""
VÁROS, AMI SZERET – Fázis 0 prototípus
Jólét-motor + kapcsolat-gráf + gazdasági körfolyamat (konzolos demo)

Futtatás:  python prototipus.py

import random
from dataclasses import dataclass, field

NEVEK = ["Anna", "Bori", "Cili", "Dániel", "Eszter", "Ferenc", "Gabi", "Hunor", "Ildi", "János"]
KESEK = ["földműves", "favágó", "kofa", "iparos"]


@dataclass
class Lako:
    """Egy falubeli. Külső és belső értékei a vonzalomszámítás alapjai."""
    nev: str
    kor: int
    kuelso: float       # 0-10
    belso: float        # 0-10
    foglalkozas: str
    jollet: float = 50.0   # 0-100
    lakas_szint: int = 1   # 1=viskó, 2=ház, 3=villa


@dataclass
class Kapcsolat:
    """Él a kapcsolat-gráfban. Egyedi kémiával és történésekkel mélyül."""
    a: int
    b: int
    ismeretseg: float = 0.0
    vonzalom: float = 0.0
    bizalom: float = 0.0
    emlekek: list = field(default_factory=list)


class Szimulacio:
    def __init__(self, n=8, seed=None):
        self.nap = 0
        self.elelem = 20.0
        self.fa = 15.0
        self.penz = 10.0
        self.lakok = []
        self.kapcsolatok = {}
        self.intim_esemenyek = []
        self.naplo = []
        if seed is not None:
            random.seed(seed)

        nevek = random.sample(NEVEK, n)
        for i in range(n):
            self.lakok.append(Lako(
                nev=nevek[i],
                kor=random.randint(18, 55),
                kuelso=round(random.uniform(3, 9), 1),
                belso=round(random.uniform(3, 9), 1),
                foglalkozas=random.choice(KESEK),
            ))
        # rejtett kémia minden párra: nem mindenki esik egymásnak
        self.kemia = {(i, j): random.uniform(0.4, 1.6)
                      for i in range(n) for j in range(i + 1, n)}

    # ---------------- VONZALOM ----------------
    def vonzalom_alap(self, i, j):
        a, b = self.lakok[i], self.lakok[j]
        alap = 0.4 * a.kuelso + 0.4 * b.belso + 0.2 * 5.0
        return (alab := alap - 5) * 10 * self.kemia[(min(i, j), max(i, j))]

    # ---------------- JÓLLÉT-MOTOR ----------------
    def jollet_szamit(self, i):
        lak = self.lakok[i]
        j = 50.0
        if self.elelem < len(self.lakok):
            j -= 30                       # éhség: kemény büntetés
        j += 5 * lak.lakas_szint          # lakhatás
        kapcs_ertek = 0.0
        for (a, b), k in self.kapcsolatok.items():
            if i in (a, b) and k.vonzalom > 50 and k.bizalom > 60:
                kapcs_ertek += (k.vonzalom + k.bizalom) / 5
        j += min(40, kapcs_ertek)         # kapcsolati elégedettség (max +40!)
        return max(0, min(100, j))

    # ---------------- ESEMÉNYEK ----------------
    def esemeny(self):
        # az ismert párok gyakrabban találkoznak -> kapcsolatok mélyülnek
        if self.kapcsolatok and random.random() < 0.7:
            kulcs = random.choice(list(self.kapcsolatok.keys()))
        else:
            i, j = random.sample(range(len(self.lakok)), 2)
            kulcs = (min(i, j), max(i, j))
            if kulcs not in self.kapcsolatok:
                self.kapcsolatok[kulcs] = Kapcsolat(a=kulcs[0], b=kulcs[1])
        k = self.kapcsolatok[kulcs]
        hely = random.choice(["piac", "kert", "kocsma", "falu_kozpont"])

        if k.ismeretseg < 20:
            k.ismeretseg += random.uniform(5, 12)
        else:
            k.vonzalom = min(100, k.vonzalom + random.uniform(5, 10) * self.kemia[kulcs])
            k.bizalom = min(100, k.bizalom + random.uniform(3, 7))
            k.emlekek.append(f"{self.nap}. nap, {hely}")
            # intim jelenet: itt kapcsolódna be a "kukkolás"
            # bizalom-lokes: magas vonzalmnal a bizalom utolerheti onmagat
            if k.vonzalom > 70 and k.bizalom < 60 and random.random() < 0.15:
                k.bizalom = min(100, k.bizalom + random.uniform(8, 15))
            if k.vonzalom > 60 and k.bizalom > 55 and random.random() < 0.35:
                self.intim_esemenyek.append((self.nap, k.a, k.b, hely))
                self.naplo.append(
                    f"  * {self.nap}. nap: {self.lakok[k.a].nev} és {self.lakok[k.b].nev} "
                    f"együtt töltötték az éjszakát ({hely}) -- KUKKOLÁSI LEHETŐSÉG!")
        k.ismeretseg = min(100, k.ismeretseg)

    # ---------------- GAZDASÁG ----------------
    def gazdasag_lep(self):
        for i, lak in enumerate(self.lakok):
            teljesitmeny = 0.5 + lak.jollet / 100      # a jólét-motor maga!
            if lak.foglalkozas == "földműves":
                self.elelem += 2.0 * teljesitmeny
            elif lak.foglalkozas == "favágó":
                self.fa += 1.5 * teljesitmeny

    # ---------------- NAPI LÉPÉS ----------------
    def lep(self):
        self.nap += 1
        if random.random() < 0.8:
            self.esemeny()
        for i in range(len(self.lakok)):
            self.lakok[i].jollet = self.jollet_szamit(i)
        self.gazdasag_lep()

    # ---------------- JELENTÉS ----------------
    def jelentes(self):
        atlag = sum(l.jollet for l in self.lakok) / len(self.lakok)
        print(f"--- {self.nap}. NAP ---  Élelem: {self.elelem:.1f}  "
              f"Fa: {self.fa:.1f}  Átlag jólét: {atlag:.1f}\n")
        print(f"{'Név':<10}{'Kor':<5}{'Külső':<7}{'Belső':<7}{'Jólét':<7}{'Foglalkozás'}")
        for l in self.lakok:
            print(f"{l.nev:<10}{l.kor:<5}{l.kuelso:<7}{l.belso:<7}{l.jollet:<7.1f}{l.foglalkozas}")
        eros = [k for k in self.kapcsolatok.values() if k.vonzalom > 40]
        print(f"\nErős kapcsolatok: {len(eros)}")
        for k in eros:
            print(f"  {self.lakok[k.a].nev} <-> {self.lakok[k.b].nev}: "
                  f"vonzalom {k.vonzalom:.0f}, bizalom {k.bizalom:.0f} ({len(k.emlekek)} emlék)")


if __name__ == "__main__":
    s = Szimulacio(n=8, seed=7)
    s.jelentes()
    print("\n" + "=" * 50 + "\n")
    for _ in range(120):
        s.lep()
    s.jelentes()
    print(f"\nIntim események száma: {len(s.intim_esemenyek)}")
    for sor in s.naplo[-6:]:
        print(sor)

function gazd(){
  S.lakok.forEach(l=>{
    const t=0.5+l.jollet/100;
    if(l.fogl==="földműves")S.elelem+=2*t;

      const aru=Math.min(S.elelem-8,2); S.elelem-=aru; S.penz+=aru*1.5; }

}

    gazd();

  render();
}
